//! The week a day falls in, counted in whole weeks from the reality's first day.
//! Spec: none, an eval fixture.
//! Twin: none, an eval fixture.
//! Invariants: days before the first day fall in weeks before week 0.

/// Why a week was refused, by its catalogue code.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum WeekError {
    /// `RUN-DIGITS`: the week left its range.
    Digits,
}

/// The week of `day`, a day number that may be negative (before the first day).
pub fn week_of(day: i64) -> Result<i64, WeekError> {
    day.checked_div_euclid(7).ok_or(WeekError::Digits)
}
