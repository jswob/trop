//! Rounding an exact ratio to a whole number, by the rounding modes of L §7.2.
//! Spec: L §7.2.
//! Twin: none, an eval fixture.
//! Invariants: the result is the multiple of 1 that the mode chooses, exactly.

use std::cmp::Ordering;

/// The rounding modes of L §7.2 this fixture implements.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Mode {
    /// The nearest; a tie goes away from zero.
    HalfUp,
    /// The nearest; a tie goes to the even multiple.
    HalfEven,
}

/// Why a rounding was refused, by its catalogue code.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum RoundError {
    /// `RUN-DIV`: the denominator isn't positive.
    Div,
    /// `RUN-DIGITS`: the result left the range of an `i64`.
    Digits,
}

/// `round(numerator / denominator, 1, mode)` (L §7.2): the whole number the mode chooses.
pub fn round(numerator: i64, denominator: i64, mode: Mode) -> Result<i64, RoundError> {
    if denominator <= 0 {
        return Err(RoundError::Div);
    }
    let floor = numerator
        .checked_div_euclid(denominator)
        .ok_or(RoundError::Digits)?;
    let rest = numerator
        .checked_rem_euclid(denominator)
        .ok_or(RoundError::Digits)?;
    let twice = rest.checked_mul(2).ok_or(RoundError::Digits)?;
    let up = match twice.cmp(&denominator) {
        Ordering::Less => false,
        Ordering::Greater => true,
        Ordering::Equal => tie_goes_up(floor, numerator, mode),
    };
    if up {
        floor.checked_add(1).ok_or(RoundError::Digits)
    } else {
        Ok(floor)
    }
}

/// Whether a tie between `floor` and `floor + 1` goes up, by the mode's rule (L §7.2).
fn tie_goes_up(floor: i64, numerator: i64, mode: Mode) -> bool {
    match mode {
        Mode::HalfUp => numerator > 0,
        Mode::HalfEven => floor.checked_rem_euclid(2) == Some(1),
    }
}
